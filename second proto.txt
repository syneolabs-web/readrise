function toggleTask(btn){
    const done=btn.classList.contains('bg-orange-500');
    if(done){
        btn.className='mt-0.5 w-5 h-5 rounded-md border border-white/12 flex items-center justify-center flex-shrink-0 hover:border-orange-500/40 transition task-check';
        btn.innerHTML='';
        const t=btn.parentElement.querySelector('p');
        t.classList.remove('line-through','text-white/30');t.classList.add('text-white/75');
    }else{
        btn.className='mt-0.5 w-5 h-5 rounded-md bg-orange-500 border border-orange-500 flex items-center justify-center flex-shrink-0 transition';
        btn.innerHTML='<span class="iconify text-white text-xs check-pop" data-icon="lucide:check"></span>';
        const t=btn.parentElement.querySelector('p');
        t.classList.add('line-through','text-white/30');t.classList.remove('text-white/75');
        showToast('Done! 🔥 Streak continues.');
    }
}

let currentFilter='all';
function setVaultFilter(t,btn){
    currentFilter=t;
    document.querySelectorAll('.tag').forEach(t=>{t.classList.remove('active');t.classList.add('text-white/35')});
    btn.classList.add('active');btn.classList.remove('text-white/35');filterVault();
}
function filterVault(){
    const s=document.getElementById('vault-search').value.toLowerCase();
    document.querySelectorAll('.insight-card').forEach(c=>{
        const mt=currentFilter==='all'||c.dataset.type===currentFilter;
        const ms=!s||c.dataset.text.includes(s);
        c.style.display=(mt&&ms)?'block':'none';
    });
}
